# 金山系资讯采集系统技术方案

## 项目概述

### 目标
建立一个自动化信息采集系统，每日09:30自动汇总过去24小时内金山系相关资讯，覆盖资本公告、公司战略合作、新品发布、WPS AI/政企产品动态、西山居游戏版本/活动、人事变动、展会采访、媒体报道等维度。

### 核心功能
- 多源信息采集（一级/二级/三级源）
- 自动去重与可信度评估
- 智能分类与标签
- 自动简报生成
- 多渠道推送（邮件/飞书/钉钉）

```python
class NewsPipeline:
    """数据处理管道"""
    
    def __init__(self):
        self.deduplicator = Deduplicator()
        self.credibility_scorer = CredibilityScorer()
        self.classifier = NewsClassifier()
        self.keyword_extractor = KeywordExtractor()
    
    def process(self, items: List[NewsItem]) -> List[ProcessedNews]:
        # 1. 清洗
        items = [self._clean(item) for item in items]
        
        # 2. 去重
        items = self.deduplicator.deduplicate(items)
        
        # 3. 可信度评分
        for item in items:
            item.credibility = self.credibility_scorer.score(item)
        
        # 4. 分类
        for item in items:
            item.category = self.classifier.classify(item)
        
        # 5. 关键词提取
        for item in items:
            item.keywords = self.keyword_extractor.extract(item)
        
        # 6. 时间过滤(仅保留24小时内)
        items = self._filter_by_time(items, hours=24)
        
        return items
```

### 3. 去重策略

```python
import hashlib
import redis
from typing import List

class Deduplicator:
    """多维度去重器"""
    
    def __init__(self, redis_client: redis.Redis):
        self.redis = redis_client
        self.url_key = "seen_urls"
        self.title_key = "seen_titles"
        self.content_sim_key = "content_signatures"
    
    def deduplicate(self, items: List[NewsItem]) -> List[NewsItem]:
        unique_items = []
        
        for item in items:
            # 维度1: URL去重(最严格)
            url_hash = hashlib.md5(item.url.encode()).hexdigest()
            if self.redis.sismember(self.url_key, url_hash):
                continue
            
            # 维度2: 标题归一化去重
            title_normalized = self._normalize_title(item.title)
            title_hash = hashlib.md5(title_normalized.encode()).hexdigest()
            if self.redis.sismember(self.title_key, title_hash):
                continue
            
            # 维度3: 内容相似度(可选,用于检测改写文)
            content_sig = self._content_signature(item.content)
            if self._similar_content_exists(content_sig):
                continue
            
            unique_items.append(item)
            
            # 记录已见
            self.redis.sadd(self.url_key, url_hash)
            self.redis.sadd(self.title_key, title_hash)
            self.redis.sadd(self.content_sim_key, content_sig)
        
        return unique_items
    
    def _normalize_title(self, title: str) -> str:
        """标题归一化:去标点、空格、统一大小写"""
        import re
        title = re.sub(r'[^\w\s]', '', title)
        title = re.sub(r'\s+', '', title)
        return title.lower().strip()
    
    def _content_signature(self, content: str) -> str:
        """生成内容指纹(用于相似度检测)"""
        import jieba
        words = list(jieba.cut(content[:200]))  # 前200字
        # SimHash或MinHash算法
        from datasketch import MinHash
        m = MinHash()
        for word in words:
            m.update(word.encode('utf8'))
        return str(m.digest())
```

### 4. 可信度评分算法

```python
class CredibilityScorer:
    """可信度评分器"""
    
    # 基础分数配置
    BASE_SCORES = {
        # 一级源 - 官方
        "cninfo": 95,           # 巨潮公告
        "hkex": 95,             # 港交所公告
        "ir_website": 90,       # IR官网
        "official_wechat": 85,  # 官方公众号
        "official_weibo": 80,   # 官方微博
        "xoyo_api": 85,         # 西山居API
        
        # 二级源 - 权威媒体
        "cls": 75,              # 财联社
        "securities_times": 75, # 证券时报
        "yicai": 70,            # 第一财经
        "news_36kr": 65,        # 36氪
        "huxiu": 65,            # 虎嗅
        "eastmoney": 60,        # 东方财富
        "xueqiu": 55,           # 雪球
        
        # 三级源 - 垂直媒体
        "gamelook": 60,         # GameLook
        "gamersky": 55,         # 游民星空
        "taptap": 55,           # TapTap
        "jiqizhixin": 60,       # 机器之心
    }
    
    def score(self, item: NewsItem) -> int:
        """计算可信度分数"""
        base = self.BASE_SCORES.get(item.source, 50)
        
        # 调整因子
        adjustments = 0
        
        # 1. 时间因子(越新越可靠)
        age_hours = (datetime.now() - item.published_at).total_seconds() / 3600
        if age_hours < 1:
            adjustments += 5
        elif age_hours > 24:
            adjustments -= 10
        
        # 2. 内容长度因子
        if len(item.content) > 500:
            adjustments += 3
        elif len(item.content) < 100:
            adjustments -= 5
        
        # 3. 来源一致性验证
        if self._verify_source(item):
            adjustments += 2
        
        return min(100, max(0, base + adjustments))
    
    def _verify_source(self, item: NewsItem) -> bool:
        """验证来源真实性"""
        # 检查URL域名是否匹配源类型
        domain_whitelist = {
            "cninfo": ["cninfo.com.cn"],
            "hkex": ["hkexnews.hk"],
            "cls": ["cls.cn"],
            # ...
        }
        # 实现验证逻辑
        return True
```

### 5. 智能分类器

```python
from enum import Enum

class NewsCategory(Enum):
    """资讯分类枚举"""
    CAPITAL = "资本公告"          # 财报、公告、减持、回购
    COOPERATION = "战略合作"      # 合作签约、生态合作
    PRODUCT = "产品动态"          # 新品发布、版本更新
    AI = "AI动态"                # WPS AI、AI产品相关
    GAME = "游戏资讯"            # 西山居游戏、金山世游
    HR = "人事变动"              # 招聘、高管变动
    MEDIA = "媒体报道"           # 采访、展会、行业报道
    POLICY = "政策合规"          # 合规、监管相关

class NewsClassifier:
    """基于规则+关键词的分类器"""
    
    RULES = {
        NewsCategory.CAPITAL: {
            "keywords": ["财报", "年报", "半年报", "季报", "营收", "净利润", 
                        "减持", "增持", "回购", "分红", "派息", "股东大会",
                        "临时公告", "内幕消息", "通函"],
            "sources": ["cninfo", "hkex", "ir_website"]
        },
        NewsCategory.COOPERATION: {
            "keywords": ["战略合作", "合作协议", "生态合作", "联合", "携手",
                        "达成", "签署", "合作框架", "伙伴关系"],
        },
        NewsCategory.PRODUCT: {
            "keywords": ["发布", "新品", "版本更新", "上线", "推出", "WPS 365",
                        "功能升级", "产品迭代", "公测", "正式上线"],
        },
        NewsCategory.AI: {
            "keywords": ["WPS AI", "AI功能", "大模型", "人工智能", "智能办公",
                        "Copilot", "AIGC", "AI助手"],
        },
        NewsCategory.GAME: {
            "keywords": ["剑网3", "尘白禁区", "解限机", "剑侠情缘", "版本更新",
                        "新资料片", "活动", "维护", "西山居", "金山世游",
                        "开服", "赛季", "联动"],
            "sources": ["xoyo_api", "wechat_game", "weibo_game"]
        },
        NewsCategory.HR: {
            "keywords": ["招聘", "诚聘", "加入", "任职", "任命", "高管",
                        "人事变动", "离职", "跳槽", "团队扩充"],
        },
        NewsCategory.MEDIA: {
            "keywords": ["采访", "专访", "报道", "展会", "峰会", "论坛",
                        "演讲", "出席", "亮相", "获奖"],
        },
    }
    
    def classify(self, item: NewsItem) -> NewsCategory:
        """分类主逻辑"""
        scores = {}
        
        for category, rules in self.RULES.items():
            score = 0
            
            # 关键词匹配
            if "keywords" in rules:
                for kw in rules["keywords"]:
                    if kw in item.title or kw in item.content:
                        score += 2
            
            # 来源匹配
            if "sources" in rules and item.source in rules["sources"]:
                score += 3
            
            scores[category] = score
        
        # 返回最高分的分类
        if scores:
            best = max(scores, key=scores.get)
            if scores[best] > 0:
                return best
        
        return NewsCategory.MEDIA  # 默认分类
```

### 6. 简报生成器

```python
from datetime import datetime
from typing import Dict, List
from jinja2 import Template

class BriefingGenerator:
    """每日简报生成器"""
    
    TEMPLATE = """
# 金山系资讯日报 - {{ date }}

## 概览
- 采集资讯总数: {{ total }}
- 一级源: {{ count_by_source['一级源'] }}
- 二级源: {{ count_by_source['二级源'] }}
- 三级源: {{ count_by_source['三级源'] }}

---

{% for category, items in sections.items() if items %}
## {{ category.value }} ({{ items|length }}条)

{% for item in items %}
### {{ loop.index }}. {{ item.title }}
- **来源**: {{ item.source }} | **可信度**: {{ item.credibility }}/100
- **时间**: {{ item.published_at.strftime('%H:%M') }}
- **链接**: [查看原文]({{ item.url }})
- **摘要**: {{ item.content[:150] }}...
{% if item.keywords %}
- **标签**: {{ item.keywords | join(', ') }}
{% endif %}

{% endfor %}
---
{% endfor %}

## 重点关注
{% for item in highlights %}
- 【{{ item.category.value }}】{{ item.title }} ({{ item.source }}, 可信度{{ item.credibility }})
{% endfor %}

---
*本简报由系统自动生成于 {{ generated_at }}*
"""
    
    def generate(self, items: List[ProcessedNews]) -> str:
        """生成简报"""
        # 按分类分组
        sections = self._group_by_category(items)
        
        # 提取高可信度重点
        highlights = [i for i in items if i.credibility >= 80]
        highlights = sorted(highlights, key=lambda x: x.credibility, reverse=True)[:5]
        
        # 统计
        total = len(items)
        count_by_source = {
            '一级源': len([i for i in items if i.credibility >= 80]),
            '二级源': len([i for i in items if 60 <= i.credibility < 80]),
            '三级源': len([i for i in items if i.credibility < 60]),
        }
        
        template = Template(self.TEMPLATE)
        return template.render(
            date=datetime.now().strftime('%Y年%m月%d日'),
            total=total,
            count_by_source=count_by_source,
            sections=sections,
            highlights=highlights,
            generated_at=datetime.now().strftime('%Y-%m-%d %H:%M:%S')
        )
    
    def _group_by_category(self, items: List[ProcessedNews]) -> Dict[NewsCategory, List]:
        """按分类分组"""
        from collections import defaultdict
        groups = defaultdict(list)
        for item in items:
            groups[item.category].append(item)
        
        # 每组按可信度排序
        for cat in groups:
            groups[cat].sort(key=lambda x: x.credibility, reverse=True)
        
        return dict(groups)
```

## 风险与应对

| 风险 | 概率 | 影响 | 应对策略 |
|------|------|------|----------|
| 微信公众号被封 | 中 | 高 | 使用第三方合规API服务，避免直接爬取 |
| 微博反爬升级 | 高 | 中 | Cookie池轮换，请求频率控制15-30秒/次 |
| 官网改版 | 中 | 中 | 监控页面结构变化，异常告警 |
| 财联社WebSocket不稳定 | 中 | 中 | 降级到REST API轮询 |
| 港交所CAPTCHA | 低 | 高 | 接入打码平台或手动处理 |
| 数据源不可用 | 低 | 高 | 多源互补设计，单源失败不影响整体 |
