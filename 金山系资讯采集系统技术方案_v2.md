# 金山系资讯采集系统技术方案 V2

## 项目概述

### 目标
建立一个自动化信息采集系统，每日09:30自动汇总过去24小时内金山系相关资讯，覆盖资本公告、公司战略合作、新品发布、WPS AI/政企产品动态、西山居游戏版本/活动、人事变动、展会采访、媒体报道等维度。

### 核心功能
- 多源信息采集（官方公告/官方资讯/媒体报道）
- 自动去重与可信度评估（三级标签体系）
- 智能五分类标签（资本/产品/合作/活动/人事）
- 时间过滤（仅保留24小时内资讯）
- 自动简报生成（每日9:30推送）
- 多渠道推送（邮件/飞书/钉钉）

```python
from playwright.sync_api import sync_playwright
from dataclasses import dataclass
from datetime import datetime, timedelta

@dataclass
class NewsItem:
    title: str
    date: str
    url: str
    source: str
    credibility_tag: str  # 【官方公告】/【官方资讯】/【媒体报道】
    category: str         # ①资本动态 ②产品动态 ③合作 ④活动 ⑤人事
    publish_time: datetime

def fetch_cninfo_688111():
    """抓取金山办公(688111)巨潮资讯公告"""
    with sync_playwright() as p:
        browser = p.chromium.launch(headless=False)
        page = browser.new_page()

        print("正在打开网页...")
        page.goto("http://www.cninfo.com.cn/new/information/topSearch/query")
        page.wait_for_load_state("networkidle")

        # 搜索股票代码
        print("正在搜索股票 688111...")
        page.fill('input[type="text"]', "688111")
        page.press('input[type="text"]', "Enter")

        # 等待搜索结果表格加载
        print("等待搜索结果...")
        page.wait_for_selector('.el-table .el-table__body-wrapper', timeout=30000)
        page.wait_for_timeout(2000)

        # 提取公告数据
        print("正在提取数据...")
        announcements = page.evaluate('''() => {
            let rows = document.querySelectorAll('.el-table__body-wrapper .el-table__row');
            if (rows.length === 0) {
                rows = document.querySelectorAll('.el-table tbody tr');
            }
            if (rows.length === 0) {
                rows = document.querySelectorAll('table tr');
            }

            const data = [];
            rows.forEach(row => {
                const cells = row.querySelectorAll('td');
                if (cells.length >= 2) {
                    const texts = [];
                    cells.forEach(c => texts.push(c.textContent?.trim() || ''));

                    // 找到日期（格式如 2026-07-29）
                    const dateMatch = texts.find(t => /\d{4}-\d{2}-\d{2}/.test(t));
                    // 找到最长的文本作为标题
                    const title = texts.filter(t => t && !/\d{4}-\d{2}-\d{2}/.test(t))
                                      .sort((a,b) => b.length - a.length)[0] || '';

                    // 找到详情链接
                    const link = row.querySelector('a')?.href || '';

                    data.push({
                        title: title,
                        date: dateMatch || '',
                        url: link
                    });
                }
            });
            return data;
        }''')

        browser.close()

        # 转换为NewsItem对象
        items = []
        for ann in announcements:
            if ann['date'] and ann['title']:
                items.append(NewsItem(
                    title=ann['title'],
                    date=ann['date'],
                    url=ann['url'],
                    source='巨潮资讯网',
                    credibility_tag='【官方公告】',
                    category=auto_classify(ann['title']),
                    publish_time=datetime.strptime(ann['date'], '%Y-%m-%d')
                ))

        return items
```

## 资讯分类标签系统（五大类）

```python
class NewsCategory:
    """资讯五分类"""
    CAPITAL = "资本动态"      # 公告、融资、股权、业绩、调研
    PRODUCT = "产品动态"      # WPS新功能、WPS AI迭代、游戏新版本/维护
    COOPERATION = "市场&政企合作"  # 签约、行业展会、战略合作
    EVENT = "活动IP"          # 游戏线下活动、联动、赛事、发布会
    HR = "人事&其他声明"      # 人事变动、其他声明

def auto_classify(title: str) -> str:
    """基于关键词的自动分类"""
    title = title.lower()

    # ①资本动态
    capital_keywords = ['公告', '财报', '年报', '半年报', '季报', '业绩', '业绩预告',
                       '营收', '净利润', '利润', '分红', '派息', '回购', '减持',
                       '增持', '股权', '融资', 'ipo', '定增', '股东大会', '调研',
                       '投资者关系', '路演']

    # ②产品动态
    product_keywords = ['wps', 'ai', '版本更新', '新功能', '上线', '发布', '迭代',
                       '剑网3', '尘白禁区', '解限机', '版本', '更新', '维护', '开服',
                       '赛季', '资料片', '新英雄', '新角色']

    # ③市场&政企合作
    cooperation_keywords = ['签约', '合作', '战略', '战略合作', '生态', '携手',
                           '联合', '伙伴关系', '政企', '中标', '采购', '合同']

    # ④活动IP
    event_keywords = ['活动', '联动', '赛事', '比赛', '发布会', '展会', 'cj',
                     'chinajoy', '线下', '见面会', '周年庆', '庆典', '嘉年华']

    # ⑤人事&其他
    hr_keywords = ['人事', '任命', '任职', '高管', '离职', '跳槽', '加入',
                  '招聘', '诚聘', '声明', '澄清', '说明']

    scores = {
        NewsCategory.CAPITAL: sum(1 for kw in capital_keywords if kw in title),
        NewsCategory.PRODUCT: sum(1 for kw in product_keywords if kw in title),
        NewsCategory.COOPERATION: sum(1 for kw in cooperation_keywords if kw in title),
        NewsCategory.EVENT: sum(1 for kw in event_keywords if kw in title),
        NewsCategory.HR: sum(1 for kw in hr_keywords if kw in title),
    }

    best = max(scores, key=scores.get)
    return best if scores[best] > 0 else NewsCategory.CAPITAL
```

## 数据管道处理流程

### 1. 时间过滤（仅保留24小时内）

```python
def filter_by_time(items: List[NewsItem], hours: int = 24) -> List[NewsItem]:
    """保留指定时间范围内的资讯"""
    cutoff_time = datetime.now() - timedelta(hours=hours)

    filtered = []
    for item in items:
        try:
            # 支持多种日期格式
            if isinstance(item.publish_time, str):
                for fmt in ['%Y-%m-%d', '%Y-%m-%d %H:%M', '%Y-%m-%d %H:%M:%S']:
                    try:
                        item.publish_time = datetime.strptime(item.publish_time, fmt)
                        break
                    except:
                        continue

            if item.publish_time and item.publish_time >= cutoff_time:
                filtered.append(item)
        except:
            continue

    return filtered
```

### 2. 去重合并（同一事件只保留一条）

```python
import hashlib
from typing import List, Dict
import jieba

class Deduplicator:
    """多维度去重器"""

    def __init__(self):
        self.seen_urls = set()
        self.seen_title_hashes = set()

    def normalize_title(self, title: str) -> str:
        """标题归一化：去标点、去空格、统一大小写"""
        import re
        # 去标点
        title = re.sub(r'[^\w\s]', '', title)
        # 去空格
        title = re.sub(r'\s+', '', title)
        # 统一小写
        return title.lower().strip()

    def extract_keywords(self, title: str) -> set:
        """提取标题核心关键词用于相似度比较"""
        # 使用jieba分词
        words = set(jieba.cut(title))
        # 过滤停用词和短词
        stopwords = {'的', '了', '和', '是', '在', '关于', '公司', '金山', '办公'}
        return words - stopwords

    def is_duplicate(self, item: NewsItem) -> bool:
        """判断是否为重复资讯"""
        # 维度1：URL去重（最严格）
        url_hash = hashlib.md5(item.url.encode()).hexdigest()
        if url_hash in self.seen_urls:
            return True

        # 维度2：标题归一化去重
        norm_title = self.normalize_title(item.title)
        title_hash = hashlib.md5(norm_title.encode()).hexdigest()
        if title_hash in self.seen_title_hashes:
            return True

        # 维度3：关键词相似度检测（同源报道）
        # 如果来源相同且标题相似度高，视为重复
        current_keywords = self.extract_keywords(item.title)
        for seen_hash in self.seen_title_hashes:
            # 简化的相似度检测
            if self._title_similarity(norm_title, seen_hash) > 0.8:
                return True

        # 记录已见
        self.seen_urls.add(url_hash)
        self.seen_title_hashes.add(title_hash)
        return False

    def _title_similarity(self, t1: str, t2_hash: str) -> float:
        """计算标题相似度（简化版）"""
        # 实际实现需要还原title_hash到原始标题或维护映射
        return 0.0

    def deduplicate(self, items: List[NewsItem]) -> List[NewsItem]:
        """执行去重"""
        unique_items = []
        for item in items:
            if not self.is_duplicate(item):
                unique_items.append(item)
        return unique_items
```

## 简报生成器（最终输出格式）

### 输出格式规范

```python
from dataclasses import dataclass
from typing import List
from datetime import datetime

@dataclass
class BriefingItem:
    title: str              # 标题
    source: str             # 来源（巨潮资讯/微信公众号/财联社等）
    publish_time: str       # 发布时间（如：2026-07-29 10:30）
    summary: str            # 摘要（100-150字）
    url: str                # 原文链接
    credibility_tag: str    # 【官方公告】/【官方资讯】/【媒体报道】
    category: str           # ①/②/③/④/⑤

class BriefingGenerator:
    """每日简报生成器"""

    def generate(self, items: List[BriefingItem], date: str = None) -> str:
        """生成Markdown格式简报"""
        if date is None:
            date = datetime.now().strftime('%Y年%m月%d日')

        # 按分类和可信度排序
        items = self._sort_items(items)

        # 按分类分组
        grouped = self._group_by_category(items)

        md = f"""# 金山系资讯日报 - {date}

## 📊 概览
- 采集时间：{datetime.now().strftime('%Y-%m-%d %H:%M')}
- 资讯总数：**{len(items)}** 条
- 官方公告：{len([i for i in items if '官方公告' in i.credibility_tag])} 条
- 官方资讯：{len([i for i in items if '官方资讯' in i.credibility_tag])} 条
- 媒体报道：{len([i for i in items if '媒体报道' in i.credibility_tag])} 条

---
"""

        # 按分类输出
        for category in ['①资本动态', '②产品动态', '③市场&政企合作', '④活动IP', '⑤人事&其他声明']:
            if category in grouped and grouped[category]:
                md += f"\n## {category}\n\n"
                for i, item in enumerate(grouped[category], 1):
                    md += f"""### {i}. {item.credibility_tag} {item.title}

- **来源**：{item.source}
- **时间**：{item.publish_time}
- **链接**：[查看原文]({item.url})
- **摘要**：{item.summary[:150]}...

---
"""

        # 重点关注（可信度≥90）
        highlights = [i for i in items if '官方公告' in i.credibility_tag]
        if highlights:
            md += f"\n## ⭐ 重点关注\n\n"
            for item in highlights[:5]:
                md += f"- **{item.category}** | {item.title} （{item.source}）\n"

        md += f"""
---
*本简报由系统自动生成于 {datetime.now().strftime('%Y-%m-%d %H:%M:%S')}*
"""
        return md

    def _sort_items(self, items: List[BriefingItem]) -> List[BriefingItem]:
        """排序：按可信度优先级，同可信度按时间倒序"""
        credibility_order = {'【官方公告】': 0, '【官方资讯】': 1, '【媒体报道】': 2}
        return sorted(items, key=lambda x: (
            credibility_order.get(x.credibility_tag, 99),
            -datetime.strptime(x.publish_time, '%Y-%m-%d %H:%M').timestamp() if x.publish_time else 0
        ))

    def _group_by_category(self, items: List[BriefingItem]) -> Dict[str, List[BriefingItem]]:
        """按分类分组"""
        from collections import defaultdict
        groups = defaultdict(list)
        for item in items:
            groups[item.category].append(item)
        return dict(groups)
```

## 完整采集流程示例

```python
def daily_collection():
    """每日采集主流程"""

    # 1. 多源采集
    all_items = []

    # 官方公告源
    cninfo_items = fetch_cninfo_688111()  # 巨潮
    all_items.extend(cninfo_items)

    # 官方资讯源
    wechat_items = fetch_wechat_official()  # 微信公众号（API）
    all_items.extend(wechat_items)

    weibo_items = fetch_weibo_official()  # 官方微博（Playwright）
    all_items.extend(weibo_items)

    # 媒体报道源
    media_items = fetch_media_reports()  # 财经/游戏媒体
    all_items.extend(media_items)

    # 2. 时间过滤（仅保留24小时内）
    recent_items = filter_by_time(all_items, hours=24)

    # 3. 自动分类
    for item in recent_items:
        item.category = auto_classify(item.title)

    # 4. 去重
    deduplicator = Deduplicator()
    unique_items = deduplicator.deduplicate(recent_items)

    # 5. 生成摘要（可选：使用大模型或简单截取）
    for item in unique_items:
        # 简单摘要：取前150字符
        item.summary = item.title + "...【摘要内容待填充】"

    # 6. 生成简报
    generator = BriefingGenerator()
    briefing = generator.generate(unique_items)

    # 7. 保存/推送
    with open(f'daily_briefing_{datetime.now().strftime("%Y%m%d")}.md', 'w', encoding='utf-8') as f:
        f.write(briefing)

    # TODO: 推送到飞书/钉钉/邮件
    # send_to_feishu(briefing)

    return briefing
```
